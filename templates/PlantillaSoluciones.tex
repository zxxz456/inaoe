% --- PATRÓN 1: demostración por inducción -----------------------------
\problema{Método de sustitución e inducción}

\noindent Resolver, con $T(1) = 1$:
\[
  T(n) = T(n/2) + 1.
\]

Una conjetura más fuerte que $\Tht{\log n}$, y por eso más fácil de
sostener en la inducción, es
\respuesta{T(n) = \log n + 1}

\paso{Caso base:} con $n = 1$,
\[
  T(1) = 1 = \log 1 + 1.
\]

\paso{Hipótesis inductiva:} se supone que vale para $n/2$,
\[
  T(n/2) = \log(n/2) + 1.
\]

\paso{Paso inductivo:} se sustituye la hipótesis y se despeja. Cada
renglón hace una sola cosa:
\begin{align*}
  T(n) &= T(n/2) + 1      \razon{definición} \\
       &= \log(n/2) + 1 + 1 \razon{hipótesis inductiva} \\
       &= \log n - \log 2 + 2 \razon{log del cociente} \\
       &= \log n - 1 + 2    \razon{$\log 2 = 1$} \\
       &= \log n + 1.
\end{align*}

Como el término dominante es $\log n$,
\respuesta{T(n) \in \Tht{\log n}}

% --- PATRÓN 2: expansión por iteración --------------------------------
\problema{Método de iteración}

\noindent Resolver, con $T(1) = 1$:
\[
  T(n) = T(n/2) + n.
\]

\paso{Expansiones:} se sustituye la recurrencia en sí misma y no se
simplifica nada todavía, para que el patrón se vea:
\begin{align*}
  T(n) &= T(n/2) + n \\
       &= T(n/4) + \frac{n}{2} + n \\
       &= T(n/8) + \frac{n}{4} + \frac{n}{2} + n.
\end{align*}

\paso{Forma general:} después de $k$ expansiones,
\[
  T(n) = T\!\left(\frac{n}{2^k}\right)
         + n \sum_{i=0}^{k-1} \left(\frac{1}{2}\right)^{i}.
\]

\paso{Dónde se detiene:} el caso base pega cuando $n/2^k = 1$, o sea
cuando \resaltar{k vale log n}. Sustituyendo ese $k$:
\begin{align*}
  T(n) &= 1 + n \cdot
          \frac{1 - (1/2)^{\log n}}{1 - 1/2} \razon{suma geométrica} \\
       &= 1 + 2n\left(1 - \frac{1}{n}\right)
          \razon{$(1/2)^{\log n} = 1/n$} \\
       &= 1 + 2n - 2 \\
       &= 2n - 1.
\end{align*}

\begin{resultado}
  $T(n) = 2n - 1$, de donde $T(n) \in \Tht{n}$. La serie se reduce a
  la mitad en cada término, así que el primer nivel domina y la suma
  entera cuesta lo mismo que la raíz salvo constante.
\end{resultado}

% --- PATRÓN 3: traza de un algoritmo ----------------------------------
\problema{Aplicación de \textsc{Max-Heapify}}

\noindent Sobre el arreglo $[16, 4, 10, 14, 7, 9, 3, 2, 8, 1]$ con
$i = 2$.

\paso{Precondición:} los subárboles con raíz en $4$ y en $5$ ya son
max-heaps. En efecto, $[14, 2, 8]$ y $[7, 1]$ lo son.

Cada llamada compara el nodo con sus dos hijos y baja al mayor. Una
fila de la tabla por comparación, sin agrupar dos pasos en uno:

\begin{table}[H]
  \centering
  \begin{tabular}{ccccl}
    \toprule
    $i$ & $A[i]$ & $A[\mathrm{izq}]$ & $A[\mathrm{der}]$
        & \textbf{Arreglo después} \\
    \midrule
    2 & 4 & 14 & 7 & $[16, 14, 10, 4, 7, 9, 3, 2, 8, 1]$ \\
    4 & 4 & 2  & 8 & $[16, 14, 10, 8, 7, 9, 3, 2, 4, 1]$ \\
    9 & 4 & -- & -- & hoja, se detiene \\
    \bottomrule
  \end{tabular}
  \caption{Traza de \textsc{Max-Heapify} desde $i = 2$.}
  \label{tab:heapify}
\end{table}

\respuesta{[16, 14, 10, 8, 7, 9, 3, 2, 4, 1]}

\begin{ojo}
  El elemento baja hasta que ya es mayor que sus dos hijos o hasta que
  llega a una hoja, no un número fijo de niveles. Por eso el peor caso
  es la altura del montículo, $\Ord{\log n}$.
\end{ojo}
