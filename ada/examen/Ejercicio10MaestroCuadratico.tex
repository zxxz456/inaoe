\encabezadoEjercicio{Método maestro: $T(n) = 2T(n/4) + n^2$}{10}{20}

\noindent\textbf{Convenciones.}
Todos los logaritmos son en base $2$.

\noindent Utilizando el método maestro, resuelva la recurrencia
\[
  T(n) = 2T(n/4) + n^2.
\]

\paso{Parámetros:} la recurrencia ya viene en la forma
$T(n) = a\,T(n/b) + f(n)$, con $a = 2$, $b = 4$ y $f(n) = n^2$. Se
cumplen $a \ge 1$ y $b > 1$, así que el método aplica.

\paso{Potencia de comparación:} se evalúa $n^{\log_b a}$.
\begin{align*}
  \log_4 2 &= \frac{\log 2}{\log 4}
    \razon{cambio de base}\\
  &= \frac{1}{2}
    \razon{$\log 2 = 1$ y $\log 4 = 2$}\\
  n^{\log_4 2} &= n^{1/2}
    \razon{se sustituye el exponente}
\end{align*}

\paso{Comparación de $f(n)$ contra $n^{1/2}$:}
\begin{align*}
  f(n) = n^2 &= n^{1/2 + 3/2}
    \razon{$2 = \tfrac{1}{2} + \tfrac{3}{2}$}\\
  &\in \Omg{n^{\log_4 2 + \varepsilon}}
    \razon{con $\varepsilon = \tfrac{3}{2} > 0$}
\end{align*}

\paso{Condición de regularidad:} hace falta una $c < 1$ con
$a\,f(n/b) \le c\,f(n)$ para toda $n$ suficientemente grande.
\begin{align*}
  a\,f(n/b) &= 2\left(\frac{n}{4}\right)^{2}
    \razon{$a = 2$, $b = 4$, $f(n) = n^2$}\\
  &= 2 \cdot \frac{n^2}{16}
    \razon{se eleva al cuadrado}\\
  &= \frac{n^2}{8}
    \razon{se simplifica}\\
  &= \frac{1}{8}\,f(n)
    \razon{$f(n) = n^2$}
\end{align*}
La desigualdad se sostiene con $c = 1/8 < 1$ para toda $n \ge 1$.

\paso{Caso que aplica:} el caso 3, porque
$f(n) \in \Omg{n^{\log_4 2 + \varepsilon}}$ con $\varepsilon = 3/2$ y
además $f$ cumple la regularidad con $c = 1/8$. Ese caso concluye
$T(n) \in \Tht{f(n)}$.

\respuesta{T(n) \in \Tht{n^2}}
